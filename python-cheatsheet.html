Use a dictionary to dispatch operations instead of long if/elif chains. W3Schools →
Dispatcher Pattern
def dispatch(op, a, b):
    operations = {
        "add": lambda x, y: x + y,
        "sub": lambda x, y: x - y,
        "mul": lambda x, y: x * y,
    }
    return operations.get(op, lambda x, y: None)(a, b)

print(dispatch("add", 10, 5))  # 15
print(dispatch("mul", 3, 4))   # 12
